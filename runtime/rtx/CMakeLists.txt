cmake_minimum_required(VERSION 3.25)
project(heterospec_rtx_native_c4 LANGUAGES C CXX)

if(NOT MSVC OR NOT WIN32)
  message(FATAL_ERROR "The preserved native worker is Windows x64/MSVC.")
endif()
if(NOT EXISTS "${NATIVE_SOURCE}/ggml/CMakeLists.txt" OR NOT EXISTS "${NATIVE_WORKER}/worker.cpp")
  message(FATAL_ERROR "Provide the verified native source and driver directories.")
endif()

set(CMAKE_CUDA_ARCHITECTURES 120a-real CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_CUDA ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA_NCCL OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_subdirectory("${NATIVE_SOURCE}" llama-build)
add_executable(rtx-c4-worker "${NATIVE_WORKER}/worker.cpp")
target_compile_features(rtx-c4-worker PRIVATE cxx_std_17)
target_compile_options(rtx-c4-worker PRIVATE /utf-8)
target_include_directories(rtx-c4-worker PRIVATE "${NATIVE_SOURCE}/vendor")
target_link_libraries(rtx-c4-worker PRIVATE llama ggml ggml-base bcrypt)
